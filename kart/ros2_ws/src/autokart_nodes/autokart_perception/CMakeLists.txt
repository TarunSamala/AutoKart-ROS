cmake_minimum_required(VERSION 3.8)
project(autokart_perception)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
set(PHASE1_CORE_PACKAGE
  "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../autonomy_stack_phase1/phase_01_perception/src/autokart_perception")
if(NOT EXISTS "${PHASE1_CORE_PACKAGE}/__init__.py")
  message(FATAL_ERROR "Authoritative Phase 1 Python package not found: ${PHASE1_CORE_PACKAGE}")
endif()
ament_python_install_package(autokart_perception PACKAGE_DIR "${PHASE1_CORE_PACKAGE}")
ament_python_install_package(autokart_perception_ros)
install(PROGRAMS scripts/perception_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_perception_adapter test/test_ros_adapter.py)
endif()
ament_package()
