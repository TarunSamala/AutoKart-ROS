cmake_minimum_required(VERSION 3.10)
project(hp60c_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT SDK_ROOT OR NOT SDK_LIB_TARGET)
  message(FATAL_ERROR "Set SDK_ROOT and SDK_LIB_TARGET")
endif()
add_library(hp60c_bridge SHARED hp60c_bridge.cpp)
target_include_directories(hp60c_bridge PRIVATE "${SDK_ROOT}/libs/include")
target_link_directories(hp60c_bridge PRIVATE "${SDK_ROOT}/libs/lib/${SDK_LIB_TARGET}")
target_link_libraries(hp60c_bridge PRIVATE AngstrongCameraSdk alg_kunlun AngKondyorArith Filt asuvc asusb turbojpeg zmq pthread)
set_target_properties(hp60c_bridge PROPERTIES BUILD_RPATH "${SDK_ROOT}/libs/lib/${SDK_LIB_TARGET}")
