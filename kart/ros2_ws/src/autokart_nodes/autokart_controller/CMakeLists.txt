cmake_minimum_required(VERSION 3.8)
project(autokart_controller)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
set(KART_SIM_PACKAGE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../kart_sim")
if(NOT EXISTS "${KART_SIM_PACKAGE}/__init__.py")
  message(FATAL_ERROR "Authoritative kart_sim Python package not found: ${KART_SIM_PACKAGE}")
endif()
ament_python_install_package(kart_sim PACKAGE_DIR "${KART_SIM_PACKAGE}")
ament_python_install_package(autokart_controller)
install(PROGRAMS scripts/controller_node DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_controller_adapter test/test_controller_adapter.py)
endif()
ament_package()
